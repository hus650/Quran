} const words = ayahText.split(' '); words.forEach(w => { if (w.trim() !== '') { const wordSpan = document.createElement('span'); wordSpan.className = 'q-word'; wordSpan.dataset.word = cleanText(w); wordSpan.innerText = w; verseSpan.className = 'verse-num'; verseSpan.innerText = ` ﴿${a.numberInSurah}﴾ `; quranBox.appendChild(verseSpan); currentSurahText += `${ayahText} ﴿${a.numberInSurah}﴾ `; }); document.getElementById('startBtn').disabled = false;
